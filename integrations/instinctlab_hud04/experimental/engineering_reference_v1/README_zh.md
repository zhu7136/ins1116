```bash
python reconstruction.py --source "$HUD04_SOURCE_MODEL" \
  --motion /path/to/kneelClimbStep1-x-0.1-ziwen-retargeted.npz \
  --terrain /path/to/50cmboxes_19_36_49.stl --output NEW_OUTPUT_DIRECTORY

  --source "$HUD04_SOURCE_MODEL" \
  --motion /path/to/kneelClimbStep1-x-0.1-ziwen-retargeted.npz \
  --terrain /path/to/50cmboxes_19_36_49.stl --output NEW_VERIFICATION.json

```
